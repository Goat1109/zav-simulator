cmake_minimum_required(VERSION 3.16)
project(robot_bundle_sender_camlidar CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)

# Draco (optional): point_codec draco needs it; without it the node still builds and refuses only that codec.
find_package(draco CONFIG QUIET)
if(TARGET draco::draco)
  set(DRACO_TARGET draco::draco)
elseif(TARGET draco_shared)
  set(DRACO_TARGET draco_shared)
elseif(TARGET draco_static)
  set(DRACO_TARGET draco_static)
endif()
if(DRACO_TARGET)
  message(STATUS "Draco found (${DRACO_TARGET}): point_codec draco enabled")
else()
  message(STATUS "Draco not found: point_codec draco disabled (apt install libdraco-dev)")
endif()

# gRPC / protobuf: CMake config (conda, source builds, newer distros) or pkg-config (Ubuntu apt libgrpc++-dev)
find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()
find_package(gRPC CONFIG QUIET)
if(gRPC_FOUND)
  set(GRPCPP_TARGET gRPC::grpc++)
  if(NOT GRPC_CPP_PLUGIN)
    get_target_property(GRPC_CPP_PLUGIN gRPC::grpc_cpp_plugin LOCATION)
  endif()
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GRPCPP REQUIRED IMPORTED_TARGET grpc++)
  set(GRPCPP_TARGET PkgConfig::GRPCPP)
  find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin REQUIRED)
endif()
if(TARGET protobuf::protoc)
  get_target_property(PROTOC_EXE protobuf::protoc LOCATION)
else()
  find_program(PROTOC_EXE protoc REQUIRED)
endif()
if(TARGET protobuf::libprotobuf)
  set(PROTOBUF_TARGET protobuf::libprotobuf)
else()
  set(PROTOBUF_TARGET ${Protobuf_LIBRARIES})
endif()

set(PROTO_FILE ${CMAKE_CURRENT_SOURCE_DIR}/proto/camlidar_bundle.proto)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${GEN_DIR})
set(GEN_SRCS ${GEN_DIR}/camlidar_bundle.pb.cc ${GEN_DIR}/camlidar_bundle.grpc.pb.cc)
add_custom_command(
  OUTPUT ${GEN_SRCS} ${GEN_DIR}/camlidar_bundle.pb.h ${GEN_DIR}/camlidar_bundle.grpc.pb.h
  COMMAND ${PROTOC_EXE} --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GEN_DIR} --grpc_out=${GEN_DIR}
          --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN} ${PROTO_FILE}
  DEPENDS ${PROTO_FILE}
  COMMENT "protoc: camlidar_bundle.proto")

add_library(robot_bundle_sender_camlidar_core STATIC src/camlidar_builder.cpp src/point_codec.cpp src/video_segment.cpp src/bundle_sender.cpp src/odom_streamer.cpp ${GEN_SRCS})
target_include_directories(robot_bundle_sender_camlidar_core PUBLIC include ${GEN_DIR})
target_link_libraries(robot_bundle_sender_camlidar_core PUBLIC Eigen3::Eigen ${GRPCPP_TARGET} ${PROTOBUF_TARGET} ZLIB::ZLIB Threads::Threads)
if(DRACO_TARGET)
  target_link_libraries(robot_bundle_sender_camlidar_core PUBLIC ${DRACO_TARGET})
  target_compile_definitions(robot_bundle_sender_camlidar_core PUBLIC HAVE_DRACO=1)
endif()
set_target_properties(robot_bundle_sender_camlidar_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_executable(bundle_sender_node src/bundle_sender_node.cpp)
target_link_libraries(bundle_sender_node robot_bundle_sender_camlidar_core)
ament_target_dependencies(bundle_sender_node rclcpp sensor_msgs nav_msgs message_filters)

install(TARGETS bundle_sender_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch proto DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  add_executable(test_camlidar_builder test/test_camlidar_builder.cpp)
  target_link_libraries(test_camlidar_builder robot_bundle_sender_camlidar_core)
  add_test(NAME test_camlidar_builder COMMAND test_camlidar_builder)
endif()

ament_package()
