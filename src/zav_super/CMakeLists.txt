cmake_minimum_required(VERSION 3.16)
project(zav_super)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs visualization_msgs mars_quadrotor_msgs ommpc_core super_planner)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/GoForward.srv")
rosidl_get_typesupport_target(super_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(super_ommpc src/super_ommpc.cpp)
target_compile_features(super_ommpc PUBLIC cxx_std_17)
target_include_directories(super_ommpc PRIVATE include ${super_planner_INCLUDE_DIRS})
target_link_libraries(super_ommpc ommpc_core::ommpc_core yaml-cpp "${super_typesupport}")
ament_target_dependencies(super_ommpc rclcpp nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs visualization_msgs mars_quadrotor_msgs)
install(TARGETS super_ommpc DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES UPSTREAM.md EPIC.md upstream_manifest.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)

  add_executable(epic_gate_test test/epic_gate.cpp)
  target_compile_features(epic_gate_test PUBLIC cxx_std_17)
  target_compile_options(epic_gate_test PRIVATE -UNDEBUG)
  target_include_directories(epic_gate_test PRIVATE include)
  add_test(NAME epic_gate COMMAND epic_gate_test)
  add_executable(trajectory_test test/trajectory.cpp)
  target_compile_features(trajectory_test PUBLIC cxx_std_17)

  target_include_directories(trajectory_test PRIVATE include)

ament_export_dependencies(rosidl_default_runtime)
ament_package()
