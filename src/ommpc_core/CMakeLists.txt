cmake_minimum_required(VERSION 3.16)
project(ommpc_core)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(osqp REQUIRED CONFIG)
add_library(ommpc_core src/controller_core.cpp)
target_compile_features(ommpc_core PUBLIC cxx_std_17)
target_include_directories(ommpc_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ommpc_core PUBLIC Eigen3::Eigen osqp::osqpstatic)
install(TARGETS ommpc_core EXPORT ommpc_coreTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(ommpc_coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 osqp)
if(BUILD_TESTING)
  enable_testing()
  add_executable(command_bounds_test test/command_bounds.cpp)
  target_include_directories(command_bounds_test PRIVATE include)
  target_link_libraries(command_bounds_test Eigen3::Eigen)
  target_compile_features(command_bounds_test PRIVATE cxx_std_17)
  target_compile_options(command_bounds_test PRIVATE -UNDEBUG)
  add_test(NAME command_bounds_test COMMAND command_bounds_test)
endif()
ament_package()
